## 项目简介

### 什么是番茄钟？
番茄钟是一种时间管理方法，由弗朗西斯科·西里洛创立。基本原理是：
- 工作时间：25分钟
- 短休息：5分钟
- 长休息：15分钟（每4个番茄钟后）

### 我们要做什么？
我们要开发一个番茄钟软件，功能包括：
- ✅ 计时器功能
- ✅ 项目管理
- ✅ 数据统计
- ✅ 云端同步
- ✅ 美观的界面

## 基础概念

### 什么是GUI？
**GUI** = Graphical User Interface（图形用户界面）
- 例子：Windows、macOS、手机APP
- 我们要做的：带按钮、输入框、显示的窗口

### 什么是Tkinter？
**Tkinter**：Python自带的GUI库
- 优点：简单易学、无需安装
- 缺点：界面相对简单
- 适用：桌面小工具、学习项目

### 什么是数据库/云存储？
**数据库**：存储数据的仓库
- 例子：Excel、SQLite
- 我们要用的：本地 JSON 文件保存数据，云端用 S3 对象存储做备份同步

### 什么是API？
**API** = Application Programming Interface（应用程序接口）
- 简单理解：函数的说明书
- 例子：`tk.Button()` 是创建按钮的API

import tkinter as tk
import time

class PomodoroApp:
    def __init__(self, root):
        self.root = root
        self.work_time = 25 * 60  # 25分钟，单位：秒
        self.current_time = self.work_time
        self.is_running = False
        
        self.create_widgets()
        self.update_timer()
    
    def create_widgets(self):
        # 时间显示
        self.time_label = tk.Label(
            self.root,
            text=self.format_time(self.current_time),
            font=("微软雅黑", 48, "bold"),
            bg="#FFFFFF",
            fg="#333333"
        )
        self.time_label.pack(pady=50)
    
    def format_time(self, seconds):
        # 将秒数转换为MM:SS格式
        mins, secs = divmod(seconds, 60)
        return f"{mins:02d}:{secs:02d}"
    
    def update_timer(self):
        # 更新计时器
        if self.is_running and self.current_time > 0:
            self.current_time -= 1
            self.time_label.config(text=self.format_time(self.current_time))
        
        # 每秒更新一次
        self.root.after(1000, self.update_timer)

root = tk.Tk()
root.title("番茄钟")
root.geometry("500x365")
root.configure(bg="#FFFFFF")

app = PomodoroApp(root)
root.mainloop()
```

def add_project(self):
    # 添加项目
    project_name = tk.simpledialog.askstring("添加项目", "请输入项目名称:")
    if project_name and project_name not in self.projects:
        self.projects.append(project_name)
        self.project_combobox.config(values=self.projects)

def delete_project(self):
    # 删除项目
    selected_project = self.project_var.get()
    if selected_project == "默认项目":
        tk.messagebox.showinfo("提示", "默认项目不能删除！")
        return
    
    if selected_project in self.projects:
        if tk.messagebox.askyesno("确认", f"确定要删除项目 '{selected_project}' 吗？"):
            self.projects.remove(selected_project)
            self.project_combobox.config(values=self.projects)
            if selected_project == self.current_project:
                self.current_project = "默认项目"
                self.project_var.set("默认项目")
```

### 5.1 使用JSON存储数据
```python
import json
import os

class PomodoroApp:
    def __init__(self, root):
        self.root = root
        self.data_file = "pomodoro_data.json"
        self.projects = ["默认项目"]
        self.current_project = "默认项目"
        self.work_time = 25 * 60
        self.current_time = self.work_time
        self.is_running = False
        self.pomodoro_count = 0
        self.today_count = 0
        
        # 加载数据
        self.load_data()
        
        self.create_widgets()
        self.update_timer()
    
    def load_data(self):
        # 加载本地数据
        if os.path.exists(self.data_file):
            with open(self.data_file, "r", encoding="utf-8") as f:
                data = json.load(f)
            if "projects" in data:
                self.projects = data["projects"]
    
    def save_data(self):
        # 保存数据
        data = {
            "projects": self.projects
        }
        with open(self.data_file, "w", encoding="utf-8") as f:
            json.dump(data, f, ensure_ascii=False, indent=2)
```

### 5.2 保存番茄钟数据

def complete_pomodoro(self):
    # 完成番茄钟
    self.is_running = False
    self.pomodoro_count += 1
    self.today_count += 1
    
    # 保存数据
    self.save_pomodoro()
    
    # 重置计时器
    self.current_time = self.work_time
    self.time_label.config(text=self.format_time(self.current_time))

def save_pomodoro(self):
    # 保存番茄钟数据
    data = self.load_data()
    today = datetime.now().strftime("%Y-%m-%d")
    
    if today not in data:
        data[today] = {}
    
    if self.current_project not in data[today]:
        data[today][self.current_project] = 0
    
    data[today][self.current_project] += 1
    
    with open(self.data_file, "w", encoding="utf-8") as f:
        json.dump(data, f, ensure_ascii=False, indent=2)

from datetime import datetime, timedelta
import tkinter.ttk as ttk

def show_weekly_day_stats(self):
    # 显示每日统计
    stats_window = tk.Toplevel(self.root)
    stats_window.title("每周每天番茄数统计")
    stats_window.geometry("600x400")
    
    # 获取本周日期
    today = datetime.now()
    week_start = today - timedelta(days=today.weekday())
    
    # 星期几名称
    weekdays = ["周一", "周二", "周三", "周四", "周五", "周六", "周日"]
    
    # 创建表格
    tree = ttk.Treeview(stats_window, columns=["date", "count"], show="headings")
    tree.heading("date", text="日期")
    tree.heading("count", text="番茄数")
    tree.pack(fill=tk.BOTH, expand=True, pady=20)
    
    # 从本地JSON获取数据（云端数据已在同步时落盘到本地）
    data = self.load_json()
    
    for i in range(7):
        date_obj = week_start + timedelta(days=i)
        date_str = date_obj.strftime("%Y-%m-%d")
        weekday_name = weekdays[i]
        display_date = f"{date_str} ({weekday_name})"
        count = sum(data.get(date_str, {}).values()) if date_str in data else 0
        tree.insert("", tk.END, values=[display_date, count])

def show_weekly_project_stats(self):
    # 显示项目统计
    stats_window = tk.Toplevel(self.root)
    stats_window.title("每周项目番茄数统计")
    stats_window.geometry("700x500")
    
    # 创建表格
    tree = ttk.Treeview(stats_window, columns=["project", "count"], show="headings")
    tree.heading("project", text="项目")
    tree.heading("count", text="番茄数")
    tree.pack(fill=tk.BOTH, expand=True, pady=20)
    
    # 加载数据
    data = self.load_data()
    project_counts = {}
    
    today = datetime.now()
    week_start = today - timedelta(days=today.weekday())
    
    for i in range(7):
        date = (week_start + timedelta(days=i)).strftime("%Y-%m-%d")
        if date in data:
            for project, count in data[date].items():
                if project not in project_counts:
                    project_counts[project] = 0
                project_counts[project] += count
    
    for project, count in project_counts.items():
        tree.insert("", tk.END, values=[project, count])

## 第七步：添加云端同步

### 7.1 开通对象存储（以阿里云 OSS 为例）
**服务地址**：https://www.aliyun.com/product/oss

**为什么用它**：阿里云 OSS、腾讯云 COS 都兼容亚马逊的 S3 协议，我们只要写一套代码就能通用。

**准备步骤**：
1. 注册并登录阿里云
2. 开通对象存储 OSS（或腾讯云 COS）
3. 创建一个 Bucket（存储桶），记下所在地域（如 cn-hangzhou）
4. 在「访问控制 → AccessKey」里创建一个 AccessKey，记下 AccessKey ID 和 AccessKey Secret

### 7.2 记录云端配置
稍后我们要把这些信息填到程序里：

- **Endpoint**（服务地址）：如 https://oss-cn-hangzhou.aliyuncs.com
- **Region**（地域）：如 cn-hangzhou（可留空）
- **Bucket**（存储桶名称）
- **Key前缀**：对象存放的目录前缀，如 `backups/`（可留空，不足结尾 `/` 时程序会自动补齐）
- **AccessKey / SecretKey**（访问密钥）
- **对象名**：默认为 pomodoro_data.json

最终云端对象的完整名字 = Key前缀 + 对象名，例如 `backups/pomodoro_data.json`。

这些配置（含密钥）会保存到本地的 cloud_config.json 文件里，请记得把它加入 .gitignore，不要提交到代码仓库。

### 7.3 添加云端同步功能
```python
import boto3
import json
import threading
import time

class PomodoroApp:
    def __init__(self, root):
        # ... 其他代码 ...

        # 云端同步设置（S3 对象存储，在 load_data 之前定义）
        self.cloud_config_file = "cloud_config.json"
        self.s3_config = {
            'endpoint': '',                     # 例如 https://oss-cn-hangzhou.aliyuncs.com
            'region': '',                       # 例如 cn-hangzhou（可留空）
            'bucket': '',                       # 存储桶名称
            'access_key': '',
            'secret_key': '',
            'prefix': '',                       # Key前缀，例如 backups/（可留空）
            'object_key': 'pomodoro_data.json'  # 云端对象名
        }
        self.cloud_enabled = False              # 是否启用云端同步
        self.s3_client = None
        self.last_sync_time = None
        self.load_cloud_config()

        # 加载数据（如果启用了云端，会先从云端拉取）
        self.load_data()

        # 创建界面
        self.create_widgets()
        self.update_clock()
        self.start_cloud_sync()

    def load_cloud_config(self):
        # 读取本地保存的云端配置
        if os.path.exists(self.cloud_config_file):
            try:
                with open(self.cloud_config_file, "r", encoding="utf-8") as f:
                    saved = json.load(f)
                for key in self.s3_config:
                    if key in saved:
                        self.s3_config[key] = saved[key]
                self.cloud_enabled = bool(saved.get("enabled", False))
            except Exception as e:
                print(f"读取云端配置失败: {str(e)}")

    def save_cloud_config(self):
        # 保存云端配置到本地文件
        try:
            data = dict(self.s3_config)
            data["enabled"] = self.cloud_enabled
            with open(self.cloud_config_file, "w", encoding="utf-8") as f:
                json.dump(data, f, ensure_ascii=False, indent=2)
        except Exception as e:
            print(f"保存云端配置失败: {str(e)}")

    def _build_s3_client(self, endpoint, region, access_key, secret_key):
        # 按传入参数创建 S3 客户端（兼容 OSS/COS 等 S3 协议服务）
        from botocore.client import Config
        return boto3.client(
            's3',
            endpoint_url=endpoint or None,
            region_name=region or None,
            aws_access_key_id=access_key or None,
            aws_secret_access_key=secret_key or None,
            # COS 要求使用虚拟主机风格域名
            config=Config(s3={'addressing_style': 'virtual'})
        )

    def _cloud_object_key(self):
        # 完整对象名 = Key前缀 + 对象名
        prefix = self.s3_config.get('prefix', '') or ''
        if prefix and not prefix.endswith('/'):
            prefix += '/'
        return prefix + (self.s3_config.get('object_key') or 'pomodoro_data.json')

    def get_s3_client(self):
        # 复用 S3 客户端
        if self.s3_client is None:
            self.s3_client = self._build_s3_client(
                self.s3_config.get('endpoint', ''),
                self.s3_config.get('region', ''),
                self.s3_config.get('access_key', ''),
                self.s3_config.get('secret_key', '')
            )
        return self.s3_client

    def init_cloud(self):
        # 校验配置并确认存储桶可访问
        if not self.s3_config.get('bucket'):
            raise Exception("请先填写 Bucket")
        client = self.get_s3_client()
        client.list_objects_v2(Bucket=self.s3_config['bucket'], MaxKeys=1)

    def upload_cloud_data(self, data):
        # 把整份数据作为单个对象上传到 S3
        client = self.get_s3_client()
        body = json.dumps(data, ensure_ascii=False, indent=2).encode('utf-8')
        client.put_object(
            Bucket=self.s3_config['bucket'],
            Key=self._cloud_object_key(),
            Body=body,
            ContentType='application/json'
        )

    def download_cloud_data(self):
        # 从 S3 下载整份数据；对象不存在时返回 None
        from botocore.exceptions import ClientError
        client = self.get_s3_client()
        try:
            obj = client.get_object(Bucket=self.s3_config['bucket'], Key=self._cloud_object_key())
            return json.loads(obj['Body'].read().decode('utf-8'))
        except ClientError as e:
            code = e.response.get('Error', {}).get('Code', '')
            if code in ('NoSuchKey', 'NoSuchBucket', '404'):
                return None
            raise

    def upload_local_to_cloud(self):
        # 把本地 JSON 同步到云端
        if not self.cloud_enabled:
            return
        try:
            data = self.load_json()
            self.upload_cloud_data(data)
        except Exception as e:
            print(f"同步本地数据到云端失败: {str(e)}")

    def load_data(self):
        # 如果启用云端同步，先把云端数据拉取到本地
        if self.cloud_enabled:
            try:
                cloud_data = self.download_cloud_data()
                if cloud_data:
                    self.save_json(cloud_data)
                else:
                    # 云端还没有数据，把本地数据推上去
                    self.upload_local_to_cloud()
            except Exception as e:
                print(f"从云端加载数据失败: {str(e)}")

        # 无论是否启用云端，项目列表都从本地 JSON 读取
        data = self.load_json()
        if "projects" in data and data["projects"]:
            self.projects = data["projects"]
        if "默认项目" not in self.projects:
            self.projects.insert(0, "默认项目")

    def start_cloud_sync(self):
        # 启动云端同步线程
        def sync_task():
            while True:
                if self.cloud_enabled:
                    # 把本地数据推送到云端
                    self.upload_local_to_cloud()
                    # 更新今日番茄数显示
                    try:
                        today_count = self.get_today_pomodoro_count()
                        if today_count > 0:
                            self.today_count = today_count
                            self.today_label.config(text=f"今日番茄数: {'🍅' * today_count}")
                        else:
                            self.today_count = 0
                            self.today_label.config(text="今日番茄数: 暂无")
                    except Exception as e:
                        print(f"获取今日番茄数失败: {str(e)}")
                time.sleep(60)  # 每分钟同步一次

        sync_thread = threading.Thread(target=sync_task, daemon=True)
        sync_thread.start()
```

### 7.4 云端设置窗口与自动同步
在界面右上角放一个"云端"按钮，点击后打开"云端设置"窗口（`show_cloud_settings`），里面可以填写 Endpoint、Region、Bucket、Key前缀、AccessKey、SecretKey、对象名，并勾选"启用云端同步"：

- **测试连接**：`test_s3_connection()` 会调用 `list_objects_v2` 验证配置是否正确。
- **保存配置**：`save_s3_config()` 把配置写入本地 cloud_config.json，并重建 S3 客户端。
- **启用/关闭同步**：勾选框绑定 `toggle_cloud_sync()`，启用时会立即把本地数据推送到云端。

启用云端后，只要数据有变化，就遵循"先改本地、再上传整份 JSON"：

```python
def save_pomodoro(self):
    # ... 读取本地 JSON、累加番茄数 ...
    self.save_json(data)

    # 同步到云端（S3）
    if self.cloud_enabled:
        self.upload_local_to_cloud()
```

`save_projects()`（增删项目）、`clear_today_data()`（今日清零）也是同样的"改本地 → 上传"逻辑。后台线程 `start_cloud_sync()` 每 60 秒把本地数据推送一次，并刷新"今日番茄数"。

## 常见问题

### Q1: 程序运行后立即关闭？
**原因**：可能是代码错误或缺少依赖
**解决**：
1. 检查控制台错误信息
2. 确认所有依赖已安装
3. 检查Python版本是否正确

### Q2: 界面显示乱码？
**原因**：字体不支持中文
**解决**：
```python
# 使用支持中文的字体
font=("微软雅黑", 12)
```

### Q3: 云端连接失败？
**原因**：对象存储配置错误或网络不通
**解决**：
1. 检查 Endpoint、Region、Bucket、AccessKey、SecretKey 是否填写正确
2. 确认存储桶存在且当前 AccessKey 有读写权限
3. 腾讯云 COS 等要求使用虚拟主机风格域名（代码已通过 `Config(s3={'addressing_style': 'virtual'})` 处理），若报 PathStyleDomainForbidden 请确认此项
4. 检查本机网络和防火墙设置

### Q4: 打包后程序无法运行？
**原因**：缺少依赖文件
**解决**：
1. 检查spec文件中的datas配置
2. 确保所有资源文件都包含
3. 重新打包

### Q5: 如何调试程序？
**方法**：在代码中添加print语句
```python
def complete_pomodoro(self):

    print(f"当前时间: {self.current_time}")
    # ... 其他代码
```
